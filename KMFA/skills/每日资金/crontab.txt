# Independent daily-funds cloud schedule.  No host scheduler, existing skill,
# event trigger, model or agent participates in these commands.
# `bootstrap-dws-auth` is deliberately absent: it is a one-time protected
# cloud-terminal command, never an unattended source of device codes.
SHELL=/bin/sh
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

*/15 * * * * root /opt/daily-funds/scripts/run_cron_job.sh poll >> /var/log/daily-funds/cron.log 2>&1
* * * * * root /opt/daily-funds/scripts/run_cron_job.sh auth-probe >> /var/log/daily-funds/cron.log 2>&1
0 * * * * root /opt/daily-funds/scripts/run_cron_job.sh keepalive >> /var/log/daily-funds/cron.log 2>&1
# Start each bounded historical batch five minutes after a live poll.  It has
# its own long lease, so a slow sparse-Git readback cannot postpone current-day
# ingestion; Git writes remain protected by the existing single-writer lock.
5,20,35,50 * * * * root /opt/daily-funds/scripts/run_cron_job.sh backfill --max-days 14 >> /var/log/daily-funds/cron.log 2>&1
30 3 * * * root /opt/daily-funds/scripts/run_cron_job.sh observer >> /var/log/daily-funds/cron.log 2>&1
0 */6 * * * root /opt/daily-funds/scripts/run_cron_job.sh r2-guard >> /var/log/daily-funds/cron.log 2>&1
10 4 * * * root /opt/daily-funds/scripts/run_cron_job.sh cold-backup >> /var/log/daily-funds/cron.log 2>&1
20 5 * * * root /opt/daily-funds/scripts/run_cron_job.sh raw-archive-audit >> /var/log/daily-funds/cron.log 2>&1
45 5 * * * root /opt/daily-funds/scripts/run_cron_job.sh runtime-audit >> /var/log/daily-funds/cron.log 2>&1
0 5 1 * * root /opt/daily-funds/scripts/run_cron_job.sh restore-drill >> /var/log/daily-funds/cron.log 2>&1
